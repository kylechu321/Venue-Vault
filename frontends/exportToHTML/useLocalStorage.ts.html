useLocalStorage.ts 
import {useEffect, useState, SetStateAction, useRef} from "react";

//https://github.com/rmit-fsd-2026-s1/lectorial04/blob/main/example4/src/hooks/useLocalStorage.ts
export function useLocalStorage<T>(
    key: string,
    initialValue: T
): [T, (value: SetStateAction<T>) => void] {
    const isFirstMount = useRef(true);

    // read once at start
    const [storedValue, setStoredValue] = useState<T>(() => {
        if (typeof window === "undefined") {
            // SSR guard — localStorage doesn't exist on the server
            return initialValue;
        }
        try {
            const item = window.localStorage.getItem(key);
            // item !== null distinguishes "key not set" from "key set to null"
            return item !== null ? (JSON.parse(item) as T) : initialValue;
        } catch (error) {
            console.error(`useLocalStorage: failed to read key "${key}"`, error);
            return initialValue;
        }
    });

    // if key changes, load that key's data
    useEffect(() => {
        if (isFirstMount.current) {
            isFirstMount.current = false;
            return;
        }
        if (typeof window === "undefined") return;
        try {
            window.localStorage.setItem(key, JSON.stringify(storedValue));
        } catch (error) {
            console.error(`useLocalStorage: failed to write key "${key}"`, error);
        }
    }, [key, storedValue]);

    const setValue = (value: SetStateAction<T>) => {
        setStoredValue(value);
    };

    return [storedValue, setValue];
}
